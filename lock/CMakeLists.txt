cmake_minimum_required(VERSION 3.20)
project(OmarchyLMIPin LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2)
add_executable(omarchy-lmi-pin-setup pin-setup.cpp)
qt_add_resources(omarchy-lmi-pin-setup pin-setup PREFIX "/" FILES PinSetup.qml)
target_link_libraries(omarchy-lmi-pin-setup PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::QuickControls2)
target_compile_options(omarchy-lmi-pin-setup PRIVATE -Wall -Wextra -Werror -fstack-protector-strong)
target_link_options(omarchy-lmi-pin-setup PRIVATE -Wl,-z,relro,-z,now,-z,noexecstack)
add_executable(verify-pin verify-pin.c)
set_target_properties(verify-pin PROPERTIES AUTOMOC OFF)
target_compile_options(verify-pin PRIVATE -Wall -Wextra -Werror -O2 -fstack-protector-strong -D_FORTIFY_SOURCE=3)
target_link_options(verify-pin PRIVATE -Wl,-z,relro,-z,now,-z,noexecstack)
target_link_libraries(verify-pin PRIVATE crypt)
